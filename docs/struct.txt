│   │   └── user.py
│   │
│   ├── schemas/          # Pydantic схемы (валидация входных/выходных данных)

│   │
│   └── services/         # Бизнес-логика приложения

│   ├── schemas/            # Схемы сериализации и валидации данных (Pydantic, Marshmallow)
│   ├── services/           # Бизнес-логика приложения
│   ├── crud/               # Операции с базой данных (Create, Read, Update, Delete)
│   └── utils/              # Вспомогательные функции, хелперы и утилиты
├── tests/                  # Автоматизированные тесты (pytest)
│   ├── __init__.py
│   ├── conftest.py         # Глобальные фикстуры для тестов
│   └── api/                # Зеркальная структура для тестирования API-роутов
├── migrations/             # Скрипты миграции базы данных (Alembic)

├── .env.example            # Шаблон переменных окружения без секретных данных
├── .gitignore              # Исключения для системы контроля версий
├── pyproject.toml          # Управление зависимостями и настройки линтеров (Poetry, ruff)
├── requirements.txt        # Стандартный файл со списком зависимостей (если используется pip)
├── Dockerfile              # Инструкции для сборки Docker-образа приложения
├── docker-compose.yml      # Оркестрация для запуска связанных сервисов (PostgreSQL, Redis, Celery)
└── README.md               # Главная документация проекта и инструкции по запуску
